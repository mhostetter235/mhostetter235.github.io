SELECT p.player_name,
       t.abbr_name AS team,
       ps.points
FROM player_stats ps
JOIN players p ON p.id = ps.player_id
JOIN teams   t ON t.id = ps.team_id
JOIN seasons s ON s.id = ps.season_id
WHERE s.season_name = '2023-24'
ORDER BY ps.points DESC
LIMIT 5;
